//LED blinking in baremetal using QCC744M

code:

#include <stdint.h>

#define GPIO_CFG1 (*(volatile uint32_t *)0x200008C8)

void delay(volatile uint32_t d)
{
    while(d--);
}

int main(void)
{
    uint32_t reg;

    /* Read current GPIO1 config */
    reg = GPIO_CFG1;

    /* Function select = SWGPIO (11) */
    reg &= ~(0x1F << 8);
    reg |=  (0x0B << 8);

    /* Mode = 00 (normal output value mode) */
    reg &= ~(3 << 30);

    /* Output enable */
    reg |= (1 << 6);

    /* Drive strength = 3 (strong drive) */
    reg &= ~(3 << 2);
    reg |=  (3 << 2);

    /* Disable:
       - Input enable (bit 0)
       - Pull-up (bit 4)
       - Pull-down (bit 5)
       - Schmitt trigger (bit 1)
    */
    reg &= ~((1 << 0) | (1 << 1) | (1 << 4) | (1 << 5));

    /* Write back */
    GPIO_CFG1 = reg;

    while(1)
    {
        /* GPIO1 HIGH */
        GPIO_CFG1 |= (1 << 24);
        delay(30000000);

        /* GPIO1 LOW */
        GPIO_CFG1 &= ~(1 << 24);
        delay(30000000);
    }
}
