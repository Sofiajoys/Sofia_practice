//code for RTC

#include "qcc74x_i2c.h"
#include "qcc74x_mtimer.h"
#include "board.h"
#include <stdio.h>

#define DS1307_ADDR  0x68

struct qcc74x_device_s *i2c0;

uint8_t dec_to_bcd(uint8_t val)
{
    return ((val / 10) << 4) | (val % 10);
}


uint8_t bcd_to_dec(uint8_t val)
{
    return ((val >> 4) * 10) + (val & 0x0F);
}


void ds1307_set_time(uint8_t hh, uint8_t mm, uint8_t ss)
{
    uint8_t buf[4];
    struct qcc74x_i2c_msg_s msg;

    buf[0] = 0x00;                  // Start at seconds register
    buf[1] = dec_to_bcd(ss) & 0x7F; // Clear CH bit
    buf[2] = dec_to_bcd(mm);
    buf[3] = dec_to_bcd(hh);        // 24-hour mode

    msg.addr   = DS1307_ADDR;
    msg.flags  = 0;
    msg.buffer = buf;
    msg.length = 4;

    qcc74x_i2c_transfer(i2c0, &msg, 1);

}


void ds1307_get_time(uint8_t *hh, uint8_t *mm, uint8_t *ss)
{
    uint8_t reg = 0x00;
    uint8_t data[3];
    struct qcc74x_i2c_msg_s msgs[2];

    
    msgs[0].addr   = DS1307_ADDR;
    msgs[0].flags  = I2C_M_NOSTOP;
    msgs[0].buffer = &reg;
    msgs[0].length = 1;

    
    msgs[1].addr   = DS1307_ADDR;
    msgs[1].flags  = I2C_M_READ;
    msgs[1].buffer = data;
    msgs[1].length = 3;

    qcc74x_i2c_transfer(i2c0, msgs, 2);

    *ss = bcd_to_dec(data[0] & 0x7F);
    *mm = bcd_to_dec(data[1]);
    *hh = bcd_to_dec(data[2] & 0x3F);
}
void ds1307_start_oscillator(void)
{
    uint8_t reg = 0x00;
    uint8_t sec;
    struct qcc74x_i2c_msg_s msgs[2];

    
    msgs[0].addr   = DS1307_ADDR;
    msgs[0].flags  = I2C_M_NOSTOP;
    msgs[0].buffer = &reg;
    msgs[0].length = 1;

  
    msgs[1].addr   = DS1307_ADDR;
    msgs[1].flags  = I2C_M_READ;
    msgs[1].buffer = &sec;
    msgs[1].length = 1;

    qcc74x_i2c_transfer(i2c0, msgs, 2);

    
    sec &= 0x7F;

    uint8_t buf[2] = {0x00, sec};
    struct qcc74x_i2c_msg_s msg;

    msg.addr   = DS1307_ADDR;
    msg.flags  = 0;
    msg.buffer = buf;
    msg.length = 2;

    qcc74x_i2c_transfer(i2c0, &msg, 1);
}


int main(void)
{
    uint8_t hour, min, sec;

    board_init();
    board_i2c0_gpio_init();

    i2c0 = qcc74x_device_get_by_name("i2c0");
    qcc74x_i2c_init(i2c0, 100000);

    printf("RTC TEST\r\n");

    ds1307_start_oscillator();     
    ds1307_set_time(12, 0, 0);    

    qcc74x_mtimer_delay_ms(2000);

    while (1) {
        ds1307_get_time(&hour, &min, &sec);
        printf("Time: %02d:%02d:%02d\r\n", hour, min, sec);
        qcc74x_mtimer_delay_ms(1000);
    }
}
