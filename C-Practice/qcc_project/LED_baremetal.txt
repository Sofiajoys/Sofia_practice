code(working):

#include <stdint.h>

#define GPIO_CFG1 (*(volatile uint32_t *)0x200008C8)

void delay(volatile uint32_t d)
{
    while(d--);
}

int main(void)
{
    uint32_t reg;


    reg = GPIO_CFG1;

    
    reg &= ~(0x1F << 8);
    reg |=  (0x0B << 8);

    
    reg &= ~(3 << 30);

    
    reg |= (1 << 6);

    
    reg &= ~(3 << 2);
    reg |=  (3 << 2);


    reg &= ~((1 << 0) | (1 << 1) | (1 << 4) | (1 << 5));


    GPIO_CFG1 = reg;

    while(1)
    {
        /* GPIO1 HIGH */
        GPIO_CFG1 |= (1 << 24);
        delay(30000000);

        /* GPIO1 LOW */
        GPIO_CFG1 &= ~(1 << 24);
        delay(30000000);
    }
}
