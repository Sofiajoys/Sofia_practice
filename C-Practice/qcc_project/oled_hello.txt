code:

#include <stdio.h>
#include "board.h"
#include "qcc74x_i2c.h"
#include "qcc74x_mtimer.h"

#define OLED_ADDR 0x3C

static struct qcc74x_device_s *i2c0;

/* ================= SSD1306 INIT (BUFFERED) ================= */
static void oled_init(void)
{
    uint8_t init_cmds[] = {
        0x00,
        0xAE,
        0x20, 0x00,
        0xB0,
        0xC8,
        0x00,
        0x10,
        0x40,
        0x81, 0x7F,
        0xA1,
        0xA6,
        0xA8, 0x3F,
        0xD3, 0x00,
        0xD5, 0x80,
        0xD9, 0xF1,
        0xDA, 0x12,
        0xDB, 0x40,
        0x8D, 0x14,
        0xAF
    };

    struct qcc74x_i2c_msg_s msg = {
        .addr = OLED_ADDR,
        .flags = 0,
        .buffer = init_cmds,
        .length = sizeof(init_cmds)
    };

    qcc74x_i2c_transfer(i2c0, &msg, 1);
}

/* ================= OLED CLEAR ================= */
static void oled_clear(void)
{
    struct qcc74x_i2c_msg_s msg;
    uint8_t buf[129];

    buf[0] = 0x40;
    for (int i = 1; i < 129; i++) buf[i] = 0x00;

    msg.addr = OLED_ADDR;
    msg.flags = 0;

    for (int page = 0; page < 8; page++)
    {
        uint8_t set_page[] = {0x00, (uint8_t)(0xB0 + page), 0x00, 0x10};

        msg.buffer = set_page;
        msg.length = sizeof(set_page);
        qcc74x_i2c_transfer(i2c0, &msg, 1);

        msg.buffer = buf;
        msg.length = sizeof(buf);
        qcc74x_i2c_transfer(i2c0, &msg, 1);
    }
}

/* ================= FONT: HELLO ================= */
static const uint8_t font_hello[5][5] = {
    {0x7F,0x08,0x08,0x08,0x7F}, // H
    {0x7F,0x49,0x49,0x49,0x41}, // E
    {0x7F,0x40,0x40,0x40,0x40}, // L
    {0x7F,0x40,0x40,0x40,0x40}, // L
    {0x3E,0x41,0x41,0x41,0x3E}  // O
};



/* ================= PRINT HELLO ================= */
static void oled_print_hello(void)
{
    struct qcc74x_i2c_msg_s msg;
    uint8_t buf[1 + 6 * 5];
    int idx = 1;

    buf[0] = 0x40;

    for (int c = 0; c < 5; c++) {
        for (int i = 0; i < 5; i++)
            buf[idx++] = font_hello[c][i];
        buf[idx++] = 0x00;
    }

    uint8_t set_page[] = {0x00, 0xB0, 0x00, 0x10};

    msg.addr = OLED_ADDR;
    msg.flags = 0;

    msg.buffer = set_page;
    msg.length = sizeof(set_page);
    qcc74x_i2c_transfer(i2c0, &msg, 1);

    msg.buffer = buf;
    msg.length = idx;
    qcc74x_i2c_transfer(i2c0, &msg, 1);
}

/* ================= MAIN ================= */
int main(void)
{
    board_init();
    board_i2c0_gpio_init();

    i2c0 = qcc74x_device_get_by_name("i2c0");
    qcc74x_i2c_init(i2c0, 400000);

    oled_init();
    oled_clear();
    oled_print_hello();

    while (1);
}
