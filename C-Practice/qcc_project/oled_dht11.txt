//displayed DHT11 sensor data on OLED

code:

#include <stdio.h>
#include <string.h>
#include "board.h"
#include "qcc74x_gpio.h"
#include "qcc74x_i2c.h"
#include "qcc74x_mtimer.h"

/* ================= CONFIG ================= */
#define OLED_ADDR  0x3C
#define DHT_PIN    GPIO_PIN_1

static struct qcc74x_device_s *gpio;
static struct qcc74x_device_s *i2c0;

/* ================= OLED INIT ================= */
static void oled_init(void)
{
    uint8_t cmds[] = {
        0x00,
        0xAE,
        0x20, 0x00,
        0xB0,
        0xC8,
        0x00, 0x10,
        0x40,
        0x81, 0x7F,
        0xA1,
        0xA6,
        0xA8, 0x3F,
        0xD3, 0x00,
        0xD5, 0x80,
        0xD9, 0xF1,
        0xDA, 0x12,
        0xDB, 0x40,
        0x8D, 0x14,
        0xAF
    };

    struct qcc74x_i2c_msg_s msg = {
        .addr = OLED_ADDR,
        .flags = 0,
        .buffer = cmds,
        .length = sizeof(cmds)
    };

    qcc74x_i2c_transfer(i2c0, &msg, 1);
}

/* ================= OLED CLEAR ================= */
static void oled_clear(void)
{
    struct qcc74x_i2c_msg_s msg;
    uint8_t buf[129];
    buf[0] = 0x40;
    memset(&buf[1], 0x00, 128);

    msg.addr = OLED_ADDR;
    msg.flags = 0;

    for (int page = 0; page < 8; page++) {
        uint8_t set_page[] = {0x00, (uint8_t)(0xB0 + page), 0x00, 0x10};
        msg.buffer = set_page;
        msg.length = sizeof(set_page);
        qcc74x_i2c_transfer(i2c0, &msg, 1);

        msg.buffer = buf;
        msg.length = sizeof(buf);
        qcc74x_i2c_transfer(i2c0, &msg, 1);
    }
}

/* ================= 5x7 FONT (T E M P H U M digits) ================= */
static const uint8_t font[][5] = {
    {0x7F,0x01,0x01,0x01,0x01}, // T
    {0x7F,0x49,0x49,0x49,0x41}, // E
    {0x7F,0x02,0x04,0x02,0x7F}, // M
    {0x7F,0x09,0x09,0x09,0x06}, // P
    {0x7F,0x08,0x08,0x08,0x7F}, // H
    {0x7F,0x40,0x40,0x40,0x3F}, // U
    {0x3E,0x51,0x49,0x45,0x3E}, // 0
    {0x00,0x42,0x7F,0x40,0x00}, // 1
    {0x42,0x61,0x51,0x49,0x46}, // 2
    {0x21,0x41,0x45,0x4B,0x31}, // 3
    {0x18,0x14,0x12,0x7F,0x10}, // 4
    {0x27,0x45,0x45,0x45,0x39}, // 5
    {0x3C,0x4A,0x49,0x49,0x30}, // 6
    {0x01,0x71,0x09,0x05,0x03}, // 7
    {0x36,0x49,0x49,0x49,0x36}, // 8
    {0x06,0x49,0x49,0x29,0x1E}  // 9
};

/* ================= OLED PRINT CHAR ================= */
static void oled_print_char(uint8_t page, uint8_t col, const uint8_t *ch)
{
    struct qcc74x_i2c_msg_s msg;
    uint8_t buf[8];
    buf[0] = 0x40;

    for (int i = 0; i < 5; i++) buf[i + 1] = ch[i];
    buf[6] = 0x00;

    uint8_t set_page[] = {0x00, (uint8_t)(0xB0 + page), col & 0x0F, 0x10 | (col >> 4)};

    msg.addr = OLED_ADDR;
    msg.flags = 0;

    msg.buffer = set_page;
    msg.length = sizeof(set_page);
    qcc74x_i2c_transfer(i2c0, &msg, 1);

    msg.buffer = buf;
    msg.length = 7;
    qcc74x_i2c_transfer(i2c0, &msg, 1);
}

/* ================= PRINT LABEL ================= */
static void oled_print_label(uint8_t page, const char *text)
{
    uint8_t col = 0;

    while (*text) {
        if (*text == 'T') oled_print_char(page, col, font[0]);
        else if (*text == 'E') oled_print_char(page, col, font[1]);
        else if (*text == 'M') oled_print_char(page, col, font[2]);
        else if (*text == 'P') oled_print_char(page, col, font[3]);
        else if (*text == 'H') oled_print_char(page, col, font[4]);
        else if (*text == 'U') oled_print_char(page, col, font[5]);
        col += 6;
        text++;
    }
}

/* ================= PRINT NUMBER ================= */
static void oled_print_number(uint8_t page, uint8_t col, int num)
{
    int tens = num / 10;
    int ones = num % 10;

    if (tens) {
        oled_print_char(page, col, font[6 + tens]);
        col += 6;
    }
    oled_print_char(page, col, font[6 + ones]);
}

/* ================= DHT11 READ ================= */
static int dht_read(uint8_t d[5])
{
    uint8_t i, j;
    uint32_t c;
    memset(d, 0, 5);

    qcc74x_gpio_init(gpio, DHT_PIN, GPIO_OUTPUT | GPIO_PULLUP);
    qcc74x_gpio_reset(gpio, DHT_PIN);
    qcc74x_mtimer_delay_ms(18);

    qcc74x_gpio_set(gpio, DHT_PIN);
    qcc74x_mtimer_delay_us(30);
    qcc74x_gpio_init(gpio, DHT_PIN, GPIO_INPUT | GPIO_PULLUP);

    while (qcc74x_gpio_read(gpio, DHT_PIN));
    while (!qcc74x_gpio_read(gpio, DHT_PIN));
    while (qcc74x_gpio_read(gpio, DHT_PIN));

    for (i = 0; i < 5; i++) {
        for (j = 0; j < 8; j++) {
            while (!qcc74x_gpio_read(gpio, DHT_PIN));
            c = 0;
            while (qcc74x_gpio_read(gpio, DHT_PIN)) {
                qcc74x_mtimer_delay_us(1);
                c++;
            }
            d[i] <<= 1;
            if (c > 50) d[i] |= 1;
        }
    }
    return ((d[0] + d[1] + d[2] + d[3]) == d[4]) ? 0 : -1;
}

/* ================= MAIN ================= */
int main(void)
{
    uint8_t dht[5];

    board_init();
    board_i2c0_gpio_init();

    gpio = qcc74x_device_get_by_name("gpio");
    i2c0 = qcc74x_device_get_by_name("i2c0");
    qcc74x_i2c_init(i2c0, 400000);

    oled_init();
    oled_clear();

    while (1)
    {
        if (dht_read(dht) == 0)
        {
            oled_clear();

            oled_print_label(0, "TEMP");
            oled_print_number(0, 40, dht[2]);

            oled_print_label(3, "HUM");
            oled_print_number(3, 40, dht[0]);
        }

        qcc74x_mtimer_delay_ms(2000);
    }
}
